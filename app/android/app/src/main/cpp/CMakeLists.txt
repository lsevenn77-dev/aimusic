cmake_minimum_required(VERSION 3.22.1)
project(aifect_monitor LANGUAGES C CXX)
add_library(aifect_monitor SHARED direct-monitor.cpp)
target_compile_features(aifect_monitor PRIVATE cxx_std_17)
target_compile_options(aifect_monitor PRIVATE -Wall -Wextra -Werror)
target_link_options(aifect_monitor PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(aifect_monitor PRIVATE aaudio android dl)

# Pinned, unmodified LGPL source is also shipped in res/raw for redistribution.
set(LAME_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/../res/raw/lame_source.tgz")
file(SHA256 "${LAME_ARCHIVE}" LAME_SHA)
if(NOT LAME_SHA STREQUAL "ddfe36cab873794038ae2c1210557ad34857a4b6bdc515785d1da9e175b1da1e")
 message(FATAL_ERROR "Unexpected LAME source archive")
endif()
file(ARCHIVE_EXTRACT INPUT "${LAME_ARCHIVE}" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/vendor")
set(LAME_ROOT "${CMAKE_CURRENT_BINARY_DIR}/vendor/lame-3.100")
set(LAME_NAMES VbrTag bitstream encoder fft gain_analysis id3tag lame newmdct presets psymodel quantize quantize_pvt reservoir set_get tables takehiro util vbrquantize version mpglib_interface)
set(LAME_SOURCES)
foreach(name IN LISTS LAME_NAMES)
 list(APPEND LAME_SOURCES "${LAME_ROOT}/libmp3lame/${name}.c")
endforeach()
add_library(aifect_lame SHARED ${LAME_SOURCES})
target_include_directories(aifect_lame PRIVATE "${LAME_ROOT}/include" "${LAME_ROOT}/libmp3lame")
target_compile_definitions(aifect_lame PRIVATE STDC_HEADERS HAVE_ERRNO_H HAVE_FCNTL_H HAVE_STDINT_H HAVE_INTTYPES_H HAVE_LIMITS_H HAVE_MEMCPY HAVE_STRCHR HAVE_STRERROR HAVE_FABS ieee754_float32_t=float)
target_link_libraries(aifect_lame PRIVATE m)
target_link_options(aifect_lame PRIVATE -Wl,-z,max-page-size=16384)
add_library(aifect_mp3 SHARED mp3-encoder.cpp)
target_include_directories(aifect_mp3 PRIVATE "${LAME_ROOT}/include")
target_compile_features(aifect_mp3 PRIVATE cxx_std_17)
target_compile_options(aifect_mp3 PRIVATE -Wall -Wextra -Werror)
target_link_libraries(aifect_mp3 PRIVATE aifect_lame)
target_link_options(aifect_mp3 PRIVATE -Wl,-z,max-page-size=16384)
